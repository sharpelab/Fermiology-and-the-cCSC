# Python dependencies for the CSC fermiology figure notebooks.
# Tested with Python 3.12. The raw-sweep loader (`sweep`) is vendored under
# utils/sweep/ and needs NO external lab package.
#
#   python -m venv .venv && source .venv/bin/activate   (or: uv venv)
#   pip install -r requirements.txt                     (or: uv pip install -r requirements.txt)

numpy
scipy
matplotlib
pandas
h5py
lmfit
scikit-learn
cycler

# --- colormaps ---
cmasher
cmcrameri

Pillow
svglib
reportlab

numba          # si-fig-onsager-model
ipywidgets     # ed-fig-optical (interactive image cropping)
